# Web UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a local-only FastAPI + HTMX browser UI wrapping the existing `game_graph.py`, sharing the same `SqliteSaver` as the CLI REPL.

**Architecture:** FastAPI app at `src/langgraph_adventure/web/server.py`. Each browser session = one `thread_id` (UUID4 in URL). Routes: GET `/` (redirect), GET `/play/new?theme=...` (create), GET `/play/{thread_id}` (render), POST `/play/{thread_id}/action` (resume interrupt), POST `/play/{thread_id}/undo`, POST `/play/{thread_id}/fork`. Server-rendered Jinja2 + HTMX; no JS framework. ~300 lines total.

**Tech Stack:** FastAPI 0.115+, uvicorn 0.30+, Jinja2 3.1+, existing `langgraph-adventure` package (game_graph, persistence, state).

**Spec:** `docs/superpowers/specs/2026-08-21-web-ui-design.md`

## Global Constraints

- Local-only browser UI (no auth, no multiplayer)
- Same SqliteSaver at `~/.langgraph_adventure/game.db` as CLI REPL
- Each browser session = one `thread_id` (UUID4 in URL `?thread=` or path)
- No JavaScript framework; ~20 lines vanilla JS for keyboard shortcuts
- Streaming stays deferred (phase 7); full scenes per turn
- All 3 deps already installed in dev env (verified at spec-write time: fastapi 0.118.0, Jinja2 3.1.4, uvicorn 0.37.0)
- Demo convention: `python -m langgraph_adventure.web.server` starts uvicorn on `127.0.0.1:8000`
- MOCK_LLM=1 works without `MINIMAX_API_KEY`
- No tests beyond manual (spec §8) — web UI is a thin HTML wrapper over already-tested game graph

```python
"""FastAPI server for langgraph-adventure web UI.

Local-only browser UI wrapping the existing game_graph. Each browser
session = one thread_id (UUID4 in URL). See spec §3.
"""
from __future__ import annotations

import uuid
from pathlib import Path

import uvicorn
from fastapi import FastAPI, Request
from fastapi.responses import RedirectResponse
from fastapi.staticfiles import StaticFiles
from fastapi.templating import Jinja2Templates

from langgraph_adventure.game_graph import graph
from langgraph_adventure.state import Scene, Action

WEB_DIR = Path(__file__).resolve().parent
TEMPLATES_DIR = WEB_DIR / "templates"
STATIC_DIR = WEB_DIR / "static"

app = FastAPI(title="langgraph-adventure")
app.mount("/static", StaticFiles(directory=str(STATIC_DIR)), name="static")
templates = Jinja2Templates(directory=str(TEMPLATES_DIR))


@app.get("/")
def root() -> RedirectResponse:
    return RedirectResponse(url="/play/new?theme=noir-detective", status_code=307)


@app.get("/play/new")
def play_new(theme: str = "noir-detective") -> RedirectResponse:
    """Create a new thread_id and redirect to /play/{thread_id}."""
    thread_id = str(uuid.uuid4())
    # Note: theme is passed to /play/{thread_id} via query param so the
    # initial state can be seeded.
    return RedirectResponse(
        url=f"/play/{thread_id}?theme={theme}", status_code=307
    )


@app.get("/play/{thread_id}")
def play(thread_id: str, theme: str = "noir-detective", request: Request = None):
    """Render the play page with the latest scene + actions.

    On first invocation (no checkpoint for thread_id), seeds the graph
    with the theme and lets `present_scene` produce the opening scene.
    """
    config = {"configurable": {"thread_id": thread_id}}
    state = graph.get_state(config)

    if state.next is None or not state.values:
        # First visit — seed and invoke.
        initial = {
            "messages": [],
            "current_scene": None,
            "chosen_action": None,
            "npc_dialogues": {},
            "theme": theme,
        }
        graph.invoke(initial, config=config)
        state = graph.get_state(config)

    scene = state.values.get("current_scene")
    actions = scene.actions if scene and scene.actions else []

    return templates.TemplateResponse(
        "play.html.j2",
        {"request": request, "thread_id": thread_id, "scene": scene, "actions": actions},
    )


if __name__ == "__main__":
    uvicorn.run("langgraph_adventure.web.server:app", host="127.0.0.1", port=8000, reload=False)
```

```python
from langgraph.types import Command

# ... (existing imports)

@app.post("/play/{thread_id}/action")
def play_action(thread_id: str, action: str = "", request: Request = None):
    """Resume the graph's interrupt with the chosen action. Returns HTMX partial."""
    config = {"configurable": {"thread_id": thread_id}}
    state = graph.get_state(config)

    if not state.next:
        return templates.TemplateResponse(
            "_narration.html.j2",
            {"request": request, "thread_id": thread_id, "scene": None, "actions": []},
        )

    graph.invoke(Command(resume=action), config=config)
    state = graph.get_state(config)
    scene = state.values.get("current_scene")
    actions = scene.actions if scene and scene.actions else []

    return templates.TemplateResponse(
        "_narration.html.j2",
        {"request": request, "thread_id": thread_id, "scene": scene, "actions": actions},
    )
```

```python
@app.post("/play/{thread_id}/undo")
def play_undo(thread_id: str, request: Request = None):
    """Rewind to the previous checkpoint's state values."""
    config = {"configurable": {"thread_id": thread_id}}
    history = list(graph.get_state_history(config))
    if len(history) < 2:
        # Nothing to undo.
        state = graph.get_state(config)
        scene = state.values.get("current_scene")
        actions = scene.actions if scene and scene.actions else []
        return templates.TemplateResponse(
            "_narration.html.j2",
            {"request": request, "thread_id": thread_id, "scene": scene, "actions": actions},
        )

    # history[0] is most recent. history[1] is one step back.
    prior = history[1]
    graph.update_state(config, values=prior.values)

    state = graph.get_state(config)
    scene = state.values.get("current_scene")
    actions = scene.actions if scene and scene.actions else []
    return templates.TemplateResponse(
        "_narration.html.j2",
        {"request": request, "thread_id": thread_id, "scene": scene, "actions": actions},
    )


@app.post("/play/{thread_id}/fork")
def play_fork(thread_id: str) -> RedirectResponse:
    """Create a new thread seeded from the current state's values."""
    config = {"configurable": {"thread_id": thread_id}}
    state = graph.get_state(config)
    new_thread_id = str(uuid.uuid4())
    new_config = {"configurable": {"thread_id": new_thread_id}}
    graph.update_state(new_config, values=state.values)
    return RedirectResponse(url=f"/play/{new_thread_id}", status_code=307)
```
